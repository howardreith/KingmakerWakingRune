$ErrorActionPreference = 'Stop'
$cases = [Collections.Generic.List[object]]::new()
function Get-Sha256([string] $Path) { (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant() }
function Quote([string] $Value) { "'" + $Value.Replace("'", "''") + "'" }
function Invoke-Isolated([string[]] $Arguments) {
    $previous = $ErrorActionPreference; $ErrorActionPreference = 'Continue'
    try { $records = $null | & powershell.exe -NoLogo -NoProfile -NonInteractive -ExecutionPolicy Bypass @Arguments 2>&1 } finally { $ErrorActionPreference = $previous }
    [pscustomobject]@{ ExitCode = $LASTEXITCODE; Output = ((@($records | ForEach-Object { "$_" })) -join "`n") }
}
function Invoke-Script([string] $Text) {
    $wrapped = "try { $Text } catch { [Console]::Out.WriteLine('ERROR: ' + `$_.Exception.Message); exit 1 }"
    Invoke-Isolated @('-EncodedCommand', [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($wrapped)))
}
function Get-Reason([string] $Output) {
    $line = @($Output -split "`n" | Where-Object { $_ -match '^ERROR: |must|does not|refused|rebuild|declined|not installed|not tracked|differ|cannot be established|changed during|replaced|failure' } | Select-Object -First 1)
    if ($line) { $line[0].Trim() } else { ($Output.Trim() -split "`n" | Select-Object -Last 1) }
}
function Add-Case([string] $Name, [bool] $Passed, [string] $Detail) {
    $cases.Add([ordered]@{ name = $Name; passed = $Passed; detail = $Detail })
    Write-Host ("{0} {1} :: {2}" -f $(if ($Passed) { 'PASS' } else { 'FAIL' }), $Name, $Detail)
}
function New-CommitClone([string] $To) {
    & git clone -q --no-hardlinks $Worktree $To
    & git -C $To checkout -q --detach $Commit
    if ((& git -C $To rev-parse HEAD).Trim() -cne $Commit) { throw "The clone did not select the commit under test: $To" }
    Copy-Item -LiteralPath (Join-Path $Worktree 'LocalGamePaths.props') -Destination $To
    New-Item -ItemType Directory -Path (Join-Path $To 'obj') -Force | Out-Null
    Copy-Item -LiteralPath (Join-Path $Worktree 'obj\loader-references') -Destination (Join-Path $To 'obj\loader-references') -Recurse
}
# R10-F3: a commit differing from Base in exactly one path; -Delete removes it (explicit), otherwise it holds Content.
function New-VariantCommit([string] $Repo, [string] $Base, [string] $Path, [string] $Content, [switch] $Delete, [string] $Message) {
    if ([bool]$Delete -eq $PSBoundParameters.ContainsKey('Content')) { throw 'New-VariantCommit needs exactly one of -Content and -Delete.' }
    $index = Join-Path $temp ('variant-index-' + [Guid]::NewGuid().ToString('N'))
    $env:GIT_INDEX_FILE = $index
    try {
        & git -C $Repo read-tree $Base
        if ($Delete) { & git -C $Repo update-index --force-remove -- $Path }
        else { $blob = ($Content | & git -C $Repo hash-object -w --stdin).Trim(); & git -C $Repo update-index --add --cacheinfo ('100644,' + $blob + ',' + $Path) }
        $tree = (& git -C $Repo write-tree).Trim()
    }
    finally { Remove-Item -Path Env:GIT_INDEX_FILE -ErrorAction SilentlyContinue; Remove-Item -LiteralPath $index -Force -ErrorAction SilentlyContinue }
    (& git -C $Repo -c user.name=focused -c user.email=focused@invalid commit-tree $tree -p $Base -m $Message).Trim()
}
function Get-VariantChange([string] $Repo, [string] $Base, [string] $Variant, [string] $Path) {
    $status = @(& git -C $Repo -c core.quotepath=off diff-tree -r --no-renames --name-status $Base $Variant | ForEach-Object { ("$_" -replace "`t", ' ').Trim() })
    $inTree = @(& git -C $Repo -c core.quotepath=off ls-tree --name-only $Variant -- $Path | Where-Object { "$_" -ceq $Path }).Count -eq 1
    [pscustomobject]@{ Status = ($status -join ' | '); InTree = $inTree }
}
function Measure-RetainedPair([string] $Producer) {

    $m = [regex]::Match(($v.Output -replace '\s+', ' '), 'Package SHA-256: ([0-9a-f]{64}); DLL SHA-256: ([0-9a-f]{64}); MVID: (\S+);.*source provenance: (\S+)')

    [ordered]@{ location = $Producer; zip = $zip; zipSha256 = (Get-Sha256 $zip); receipt = $receipt; receiptSha256 = (Get-Sha256 $receipt); producingCommit = [string]$rj.commit; dirty = $rj.dirty
        receiptDllSha256 = [string]$rj.dllSha256; validateExit = $v.ExitCode; dllSha256 = $(if ($m.Success) { $m.Groups[2].Value } else { $null }); dllMvid = $(if ($m.Success) { $m.Groups[3].Value } else { $null })

}

    Add-Case 'r9f4-genuine-producer-with-tracked-equals-response-file-qualifies' ($pb.ExitCode -eq 0 -and [string]$pr.commit -ceq $producerCommit -and $entries -match 'response-file:@Rules=Base\.rsp' -and $material.Output -match 'MATERIAL CaseRoot\.cs,Rules=Base\.rsp,Rules=Root\.cs' -and $genuine.ExitCode -eq 0 -and $genuine.Output -match 'source provenance: verified-current') ("build exit $($pb.ExitCode); receipt commit $($pr.commit) (producer $producerCommit); inventory entries [$entries]; comparison paths " + [regex]::Match($material.Output, 'MATERIAL \S*').Value + '; ' + [regex]::Match(($genuine.Output -replace '\s+', ' '), 'source provenance: \S+').Value)
    $helperText = [IO.File]::ReadAllText((Join-Path $base 'scripts\KwrFileIdentity.ps1'))
    $variants = @(
        [pscustomobject]@{ Name = 'casing-source-missing'; Path = 'CaseRoot.cs'; Change = 'D'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'CaseRoot.cs' -Delete -Message 'v') },
        [pscustomobject]@{ Name = 'casing-source-differing'; Path = 'CaseRoot.cs'; Change = 'M'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'CaseRoot.cs' -Content 'namespace KingmakerWakingRune.LocalOnly { internal static class CaseRootMarker { internal const int Value = 11; } }' -Message 'v') },
        [pscustomobject]@{ Name = 'equals-source-missing'; Path = 'Rules=Root.cs'; Change = 'D'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'Rules=Root.cs' -Delete -Message 'v') },
        [pscustomobject]@{ Name = 'equals-source-differing'; Path = 'Rules=Root.cs'; Change = 'M'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'Rules=Root.cs' -Content 'namespace KingmakerWakingRune.LocalOnly { internal static class EqualsRootMarker { internal const int Value = 14; } }' -Message 'v') },
        [pscustomobject]@{ Name = 'equals-response-file-differing'; Path = 'Rules=Base.rsp'; Change = 'M'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'Rules=Base.rsp' -Content '# another tracked response file' -Message 'v') },
        [pscustomobject]@{ Name = 'identity-helper-differing'; Path = 'scripts/KwrFileIdentity.ps1'; Change = 'M'; Commit = (New-VariantCommit -Repo $base -Base $producerCommit -Path 'scripts/KwrFileIdentity.ps1' -Content ($helperText + "`n# variant`n") -Message 'v') })
    foreach ($variant in $variants) {
        $change = Get-VariantChange $base $producerCommit $variant.Commit $variant.Path
        $established = $change.Status -ceq ($variant.Change + ' ' + $variant.Path) -and $change.InTree -eq ($variant.Change -cne 'D')
        $vr = Join-Path $temp "receipt-$($variant.Name).json"
        $j = Get-Content -LiteralPath $receiptPath -Raw | ConvertFrom-Json; $j.commit = $variant.Commit; $j | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $vr -Encoding UTF8
        $res = Invoke-Isolated @('-File', $validator, '-PackagePath', $candidateZip, '-ReceiptPath', $vr, '-RequireCurrentProvenance')
        Add-Case "r10f3-producer-$($variant.Name)-refused" ($established -and $res.ExitCode -ne 0 -and ($res.Output -replace '\s+', ' ') -match "material inputs differ from the receipt's commit") ("Git change from the producer: [$($change.Status)] (expected [$($variant.Change) $($variant.Path)]); in the variant tree: $($change.InTree); established: $established; " + (Get-Reason $res.Output))
    }
    $classVariant = @($variants | Where-Object { $_.Name -ceq 'casing-source-differing' })[0]
    $cr2 = Join-Path $temp 'receipt-source-classified-external.json'
    $j = Get-Content -LiteralPath $receiptPath -Raw | ConvertFrom-Json; $j.commit = $classVariant.Commit
    foreach ($e in @($j.compilerInputs | Where-Object { [string]$_.kind -ceq 'compile' -and [string]$_.name -ieq 'CaseRoot.cs' })) { $e.repositoryPath = $null }
    $j | ConvertTo-Json -Depth 8 | Set-Content -LiteralPath $cr2 -Encoding UTF8
    $views = Invoke-Script (". $(Quote (Join-Path $base 'scripts\KwrBuild.Common.ps1')); `$r = Get-Content -LiteralPath $(Quote $cr2) -Raw | ConvertFrom-Json; 'RECEIPT ' + ((@(`$r.compilerInputs | Where-Object { [string]`$_.name -ieq 'CaseRoot.cs' } | ForEach-Object { 'repositoryPath=' + [string]`$_.repositoryPath })) -join ',') + ' paths=' + ((Get-KwrInventoryRepositoryPaths `$r.compilerInputs | Where-Object { `$_ -ieq 'CaseRoot.cs' }) -join ','); 'LIVE ' + ((@(Get-KwrCompilerInputs | Where-Object { [string]`$_.kind -ceq 'compile' -and [string]`$_.name -ieq 'CaseRoot.cs' } | ForEach-Object { 'repositoryPath=' + [string]`$_.repositoryPath })) -join ',')")
    $classChange = Get-VariantChange $base $producerCommit $classVariant.Commit 'CaseRoot.cs'
    $classEstablished = $views.Output -match '(?m)^RECEIPT repositoryPath= paths=\s*$' -and $views.Output -match '(?m)^LIVE repositoryPath=CaseRoot\.cs\s*$' -and $classChange.Status -ceq 'M CaseRoot.cs' -and $classChange.InTree
    $classResult = Invoke-Isolated @('-File', $validator, '-PackagePath', $candidateZip, '-ReceiptPath', $cr2, '-RequireCurrentProvenance')
    Add-Case 'r10f2-receipt-recording-a-tracked-root-source-as-external-with-a-producer-differing-only-in-it-refused' ($classEstablished -and $classResult.ExitCode -ne 0 -and ($classResult.Output -replace '\s+', ' ') -match "material inputs differ from the receipt's commit" -and $genuine.ExitCode -eq 0) ("receipt view: " + [regex]::Match($views.Output, 'RECEIPT [^\r\n]*').Value + "; live view: " + [regex]::Match($views.Output, 'LIVE [^\r\n]*').Value + "; Git change: [$($classChange.Status)]; established: $classEstablished; " + (Get-Reason $classResult.Output) + "; genuine pair: " + [regex]::Match(($genuine.Output -replace '\s+', ' '), 'source provenance: \S+').Value)
    $decodeProbe = ". $(Quote (Join-Path $base 'scripts\KwrBuild.Common.ps1')); foreach (`$e in @(@('file-option', '/additionalfile:Rules=Base.rsp'), @('file-option', '/resource:Rules=Base.rsp,Logical=Name'), @('file-option', '/win32res:Rules=Root.cs'), @('response-file', '@Rules=Base.rsp'), @('reference', 'Rules=Base.rsp'))) { 'DECODE ' + `$e[0] + ' ' + `$e[1] + ' -> [' + ((Get-KwrInventoryRepositoryPaths @([ordered]@{ kind = `$e[0]; name = `$e[1]; sha256 = 'x' })) -join ',') + ']' }"
    $decoded = Invoke-Script $decodeProbe
    $decodeLines = @($decoded.Output -split "`n" | Where-Object { $_ -match '^DECODE ' } | ForEach-Object { $_.Trim() })
    $decodeExpected = @('DECODE file-option /additionalfile:Rules=Base.rsp -> [Rules=Base.rsp]', 'DECODE file-option /resource:Rules=Base.rsp,Logical=Name -> [Rules=Base.rsp]', 'DECODE file-option /win32res:Rules=Root.cs -> [Rules=Root.cs]', 'DECODE response-file @Rules=Base.rsp -> [Rules=Base.rsp]', 'DECODE reference Rules=Base.rsp -> []')
    Add-Case 'r9f4-inventory-decoding-keeps-literal-equals-signs-outside-reference-aliases' ($decoded.ExitCode -eq 0 -and ($decodeLines -join '|') -ceq ($decodeExpected -join '|')) ($decodeLines -join '; ')
    [IO.File]::WriteAllText($localProps, $propsBefore)
}
finally {
    Set-Location -LiteralPath (Split-Path -Parent $OutFile)
    if (Test-Path -LiteralPath $temp) { try { Remove-Item -LiteralPath $temp -Recurse -Force -ErrorAction Stop } catch { Write-Warning "temporary clone not removed: $temp ($($_.Exception.Message))" } }
    $summary.disposableRemoved = -not (Test-Path -LiteralPath $temp)
    $failed = @($cases | Where-Object { -not $_.passed })
    $summary.passed = $cases.Count - $failed.Count; $summary.total = $cases.Count; $summary.cases = @($cases)
    $summary | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath $OutFile -Encoding UTF8
    Write-Host ("RESULT {0}/{1}" -f $summary.passed, $summary.total)
}
if (@($cases | Where-Object { -not $_.passed }).Count -gt 0) { exit 1 }

$ErrorActionPreference = 'Stop'
$cases = [Collections.Generic.List[object]]::new()
function Get-Sha256([string] $Path) { (Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant() }
function Get-ShaOrMissing([string] $Path) { if (Test-Path -LiteralPath $Path) { Get-Sha256 $Path } else { 'missing' } }
function Quote([string] $Value) { "'" + $Value.Replace("'", "''") + "'" }
function Invoke-Isolated([string[]] $Arguments) {
    $previous = $ErrorActionPreference; $ErrorActionPreference = 'Continue'
    try { $records = $null | & powershell.exe -NoLogo -NoProfile -NonInteractive -ExecutionPolicy Bypass @Arguments 2>&1 } finally { $ErrorActionPreference = $previous }
    [pscustomobject]@{ ExitCode = $LASTEXITCODE; Output = ((@($records | ForEach-Object { "$_" })) -join "`n") }
}
function Invoke-Script([string] $Text) {
    $wrapped = "try { $Text } catch { [Console]::Out.WriteLine('ERROR: ' + `$_.Exception.Message); exit 1 }"
    Invoke-Isolated @('-EncodedCommand', [Convert]::ToBase64String([Text.Encoding]::Unicode.GetBytes($wrapped)))
}
function Get-Reason([string] $Output) {
    $line = @($Output -split "`n" | Where-Object { $_ -match '^ERROR: |must|does not|refused|rebuild|declined|not installed|not tracked|differ|cannot be|changed|replaced|moved|failure|written only|Build inputs' } | Select-Object -First 1)
    if ($line) { $line[0].Trim() } else { ($Output.Trim() -split "`n" | Select-Object -Last 1) }
}
function Add-Case([string] $Name, [bool] $Passed, [string] $Detail) {
    $cases.Add([ordered]@{ name = $Name; passed = $Passed; detail = $Detail })
    Write-Host ("{0} {1} :: {2}" -f $(if ($Passed) { 'PASS' } else { 'FAIL' }), $Name, $Detail)
}
function New-CommitClone([string] $To) {
    & git clone -q --no-hardlinks $Worktree $To
    & git -C $To checkout -q --detach $Commit
    if ((& git -C $To rev-parse HEAD).Trim() -cne $Commit) { throw "The clone did not select the commit under test: $To" }
    Copy-Item -LiteralPath (Join-Path $Worktree 'LocalGamePaths.props') -Destination $To
    New-Item -ItemType Directory -Path (Join-Path $To 'obj') -Force | Out-Null
    Copy-Item -LiteralPath (Join-Path $Worktree 'obj\loader-references') -Destination (Join-Path $To 'obj\loader-references') -Recurse
}
function Measure-RetainedPair([string] $Producer) {

    $zip = Join-Path $Producer "artifacts\KingmakerWakingRune-$version.zip"; $receipt = $zip + '.build.json'
    $v = Invoke-Isolated @('-File', (Join-Path $Producer 'scripts\Validate-Package.ps1'), '-PackagePath', $zip, '-ReceiptPath', $receipt, '-RequireCurrentProvenance')
    $m = [regex]::Match(($v.Output -replace '\s+', ' '), 'Package SHA-256: ([0-9a-f]{64}); DLL SHA-256: ([0-9a-f]{64}); MVID: (\S+);.*source provenance: (\S+)')
    $rj = Get-Content -LiteralPath $receipt -Raw | ConvertFrom-Json
    [ordered]@{ location = $Producer; zip = $zip; zipSha256 = (Get-Sha256 $zip); receipt = $receipt; receiptSha256 = (Get-Sha256 $receipt); producingCommit = [string]$rj.commit; dirty = $rj.dirty
        receiptDllSha256 = [string]$rj.dllSha256; validateExit = $v.ExitCode; dllSha256 = $(if ($m.Success) { $m.Groups[2].Value } else { $null }); dllMvid = $(if ($m.Success) { $m.Groups[3].Value } else { $null })
        provenance = $(if ($m.Success) { $m.Groups[4].Value } else { $null }); validatedZipSha256 = $(if ($m.Success) { $m.Groups[1].Value } else { $null }); producerStatusEntries = @(& git -C $Producer status --porcelain).Count
        measuredUtc = [DateTime]::UtcNow.ToString('o') }
}
# The actual Csc reference arguments naming Accessibility.dll, from the same SkipCompilerExecution pass the inventory uses.
function Get-AccessibilityArguments([string] $Repo) {
    $probe = ". $(Quote (Join-Path $Repo 'scripts\KwrBuild.Common.ps1')); `$id = [Guid]::NewGuid().ToString('N'); `$list = Join-Path ([IO.Path]::GetTempPath()) ('kwr-args-' + `$id + '.txt'); `$obj = Join-Path ([IO.Path]::GetTempPath()) ('kwr-args-' + `$id + '\'); `$null | & (Get-KwrMsBuild) $(Quote (Join-Path $Repo 'src\KingmakerWakingRune\KingmakerWakingRune.csproj')) /nologo /v:q /nr:false '/t:PrepareResources;Compile' /p:SkipCompilerExecution=true /p:Configuration=Release (`"/p:CompilerReferenceListPath=`" + `$list) (`"/p:IntermediateOutputPath=`" + `$obj) (`"/p:OutputPath=`" + `$obj + 'bin') | Out-Null; foreach (`$l in @(Get-Content -LiteralPath `$list | Where-Object { `$_ -like '*Accessibility*' })) { 'ARG ' + `$l }; Remove-Item -LiteralPath `$list -Force; if ([IO.Directory]::Exists(`$obj)) { [IO.Directory]::Delete(`$obj, `$true) }"
    $r = Invoke-Script $probe
    $argList = @($r.Output -split "`n" | Where-Object { $_ -match '^ARG ' } | ForEach-Object { $_.Substring(4).Trim().Replace($Repo, '<clone>') })
    [pscustomobject]@{ ExitCode = $r.ExitCode; Arguments = ($argList -join ' | ') }
}

        $sameBytes = (Get-Sha256 (Join-Path $base 'Rules=Local\Accessibility.dll')) -ceq (Get-Sha256 (Join-Path $base 'Local\Accessibility.dll'))
        $literalProps = $propsBefore.Replace('</Project>', "  <ItemGroup><Reference Include=`"Accessibility`"><HintPath>`$(MSBuildThisFileDirectory)Rules=Local\Accessibility.dll</HintPath></Reference></ItemGroup>`r`n</Project>")
        $aliasProps = $propsBefore.Replace('</Project>', "  <ItemGroup><Reference Include=`"Accessibility`"><HintPath>`$(MSBuildThisFileDirectory)Local\Accessibility.dll</HintPath><Aliases>Rules</Aliases></Reference></ItemGroup>`r`n</Project>")
        [IO.File]::WriteAllText($localProps, $literalProps)
        $cz = Join-Path $base "artifacts\KingmakerWakingRune-$version.zip"; $cr = $cz + '.build.json'
        $pb = Invoke-Isolated @('-File', (Join-Path $base 'scripts\Build.ps1'), '-SkipTests', '-ReplacePackage')
        $argsLiteral = Get-AccessibilityArguments $base
        $literalOk = Invoke-Isolated @('-File', $validator, '-PackagePath', $cz, '-ReceiptPath', $cr, '-RequireCurrentProvenance')
        [IO.File]::WriteAllText($localProps, $aliasProps)
        $argsAlias = Get-AccessibilityArguments $base
        $aliasRun = Invoke-Isolated @('-File', $validator, '-PackagePath', $cz, '-ReceiptPath', $cr, '-RequireCurrentProvenance')
        [IO.File]::WriteAllText($localProps, $literalProps)
        $restoredRun = Invoke-Isolated @('-File', $validator, '-PackagePath', $cz, '-ReceiptPath', $cr, '-RequireCurrentProvenance')
        [IO.File]::WriteAllText($localProps, $propsBefore)
        $prov = { param($r) [regex]::Match(($r.Output -replace '\s+', ' '), 'source provenance: \S+').Value + $(if ($r.ExitCode -ne 0) { ' REFUSED: ' + (Get-Reason $r.Output) } else { '' }) }
        $obs22 = "both paths tracked at producer $($producerCommit.Substring(0, 7)) with identical bytes: $sameBytes; build exit $($pb.ExitCode); literal arguments [$($argsLiteral.Arguments)]: " + (& $prov $literalOk) + "; alias arguments [$($argsAlias.Arguments)]: " + (& $prov $aliasRun) + "; literal restored: " + (& $prov $restoredRun)
        $setup = $sameBytes -and $pb.ExitCode -eq 0 -and $argsLiteral.Arguments -match '/reference:.*Rules=Local\\Accessibility\.dll' -and $argsAlias.Arguments -match '/reference:Rules=.*\\Local\\Accessibility\.dll' -and $literalOk.ExitCode -eq 0 -and $literalOk.Output -match 'verified-current'
        if ($fixed) { Add-Case 'r12f2-reference-binding-change-refused-and-restored-configuration-qualifies' ($setup -and $aliasRun.ExitCode -ne 0 -and ($aliasRun.Output -replace '\s+', ' ') -match 'Build inputs changed' -and $restoredRun.ExitCode -eq 0 -and $restoredRun.Output -match 'verified-current') $obs22 }
        else { Add-Case 'r12f2-reproduced-old-pair-verified-current-after-the-binding-changed' ($setup -and $aliasRun.ExitCode -eq 0 -and $aliasRun.Output -match 'verified-current') $obs22 }
    }
}
finally {
    Set-Location -LiteralPath (Split-Path -Parent $OutFile)
    if (Test-Path -LiteralPath $temp) { try { Remove-Item -LiteralPath $temp -Recurse -Force -ErrorAction Stop } catch { Write-Warning "temporary clone not removed: $temp ($($_.Exception.Message))" } }
    $summary.disposableRemoved = -not (Test-Path -LiteralPath $temp)
    $failed = @($cases | Where-Object { -not $_.passed })
    $summary.finishedUtc = [DateTime]::UtcNow.ToString('o')
    $summary.passed = $cases.Count - $failed.Count; $summary.total = $cases.Count; $summary.cases = @($cases)
    if (Test-Path -LiteralPath $OutFile) { throw "Refusing to overwrite an existing output: $OutFile" }
    $summary | ConvertTo-Json -Depth 6 | Set-Content -LiteralPath $OutFile -Encoding UTF8
    Write-Host ("RESULT {0}/{1}" -f $summary.passed, $summary.total)
}
if (@($cases | Where-Object { -not $_.passed }).Count -gt 0 -or $cases.Count -eq 0) { exit 1 }
